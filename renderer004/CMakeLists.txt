cmake_minimum_required(VERSION 3.20)
project(cc_replay004 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CC_SANITIZERS "Sanitize host tests" OFF)
add_library(cc_replay_core replay.cpp gpu_staging.cpp)
target_compile_options(cc_replay_core PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
if(CC_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(cc_replay_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(cc_replay host_main.cpp)
target_link_libraries(cc_replay PRIVATE cc_replay_core)
add_executable(cc_replay_tests tests.cpp)
target_link_libraries(cc_replay_tests PRIVATE cc_replay_core)
enable_testing()
add_test(NAME replay_tests COMMAND cc_replay_tests)
