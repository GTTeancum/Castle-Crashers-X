cmake_minimum_required(VERSION 3.20)
project(cc_stream005 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CC_SANITIZERS "Sanitize host tests" OFF)
if(CC_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(cc_stream stream_loader.cpp ../renderer004/replay.cpp ../renderer004/gpu_staging.cpp)
target_compile_options(cc_stream PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
add_executable(cc_stream_tests tests.cpp)
target_compile_options(cc_stream_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(cc_stream_tests PRIVATE cc_stream)
enable_testing()
add_test(NAME stream_unit COMMAND cc_stream_tests)
add_test(NAME stream_real COMMAND cc_stream_tests ${CMAKE_CURRENT_SOURCE_DIR}/../renderer004/menu.ccr ${CMAKE_CURRENT_SOURCE_DIR}/../renderer004/book.ccr)
