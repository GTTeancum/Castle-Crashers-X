cmake_minimum_required(VERSION 3.25)
project(cc_mode009 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CC_SANITIZERS "Run host sanitizer checks" OFF)
add_executable(cc_mode009_tests tests.cpp mode.cpp ../xbox-runtime/guest_memory.cpp)
target_compile_definitions(cc_mode009_tests PRIVATE CC_MODE009_TEST_SERVICE=1)
target_compile_options(cc_mode009_tests PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
if(CC_SANITIZERS)
 target_compile_options(cc_mode009_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(cc_mode009_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME mode009 COMMAND cc_mode009_tests)
