cmake_minimum_required(VERSION 3.20)
project(cc_quad006 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(CC_SANITIZERS "Enable host ASan/UBSan" OFF)
add_library(cc_quad STATIC bridge.cpp fixtures.cpp ../xbox-runtime/guest_memory.cpp)
target_compile_options(cc_quad PUBLIC -Wall -Wextra -Werror -ffp-contract=off -fno-fast-math)
if(CC_SANITIZERS)
 target_compile_options(cc_quad PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(cc_quad PUBLIC -fsanitize=address,undefined)
endif()
add_executable(cc_quad_tests host_tests.cpp)
target_link_libraries(cc_quad_tests PRIVATE cc_quad)
enable_testing()
add_test(NAME captured_quad_equivalence COMMAND cc_quad_tests ${CMAKE_CURRENT_SOURCE_DIR}/../renderer006/menu.ccq ${CMAKE_CURRENT_SOURCE_DIR}/../renderer006/book.ccq)
