cmake_minimum_required(VERSION 3.20)
project(cc_asset_probe LANGUAGES CXX)
option(CC_SANITIZERS "Build host tests with AddressSanitizer/UBSan" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(cc_asset_core STATIC asset_core/asset_reader.cpp)
target_include_directories(cc_asset_core PUBLIC asset_core)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(cc_asset_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
  if(CC_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    target_compile_options(cc_asset_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()
add_executable(cc_asset_probe asset_core/probe_main.cpp)
target_link_libraries(cc_asset_probe PRIVATE cc_asset_core)
add_executable(cc_asset_tests asset_core/tests.cpp)
target_link_libraries(cc_asset_tests PRIVATE cc_asset_core)
enable_testing()
add_test(NAME cc_asset_tests COMMAND cc_asset_tests)
