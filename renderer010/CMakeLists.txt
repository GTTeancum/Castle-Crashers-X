cmake_minimum_required(VERSION 3.20)
project(cc_owner010 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(CC_SANITIZERS "Enable ASan/UBSan" OFF)
add_subdirectory(../runtime008 runtime)
add_library(cc_scene_frame STATIC live_frame.cpp ../renderer005/stream_loader.cpp ../renderer004/replay.cpp)
target_link_libraries(cc_scene_frame PUBLIC cc_tree)
add_executable(cc_scene_preview host_frame.cpp)
target_link_libraries(cc_scene_preview PRIVATE cc_scene_frame)
add_library(cc_scene_fused STATIC ../runtime007/numeric/fused.cpp)
add_library(cc_scene_fma OBJECT ../runtime007/numeric/target_fma.cpp)
add_executable(cc_scene_preview_soft host_frame.cpp $<TARGET_OBJECTS:cc_scene_fma>)
target_link_libraries(cc_scene_preview_soft PRIVATE cc_scene_frame cc_scene_fused)
add_executable(cc_tree_tests_soft ../runtime010/host_owner_tests.cpp $<TARGET_OBJECTS:cc_scene_fma>)
target_link_libraries(cc_tree_tests_soft PRIVATE cc_tree cc_scene_fused)

if(CC_SANITIZERS)
 target_compile_options(cc_scene_fused PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_compile_options(cc_scene_fma PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

target_compile_options(cc_scene_fused PRIVATE -Wall -Wextra -Werror -ffp-contract=off -fno-fast-math)
add_executable(cc_fixture_tests ../runtime010/fixture_tests.cpp)
target_link_libraries(cc_fixture_tests PRIVATE cc_tree)
enable_testing()
add_test(NAME tree_fixture_guards COMMAND cc_fixture_tests ${CMAKE_CURRENT_SOURCE_DIR}/startup.cct ${CMAKE_CURRENT_SOURCE_DIR}/menu.cct ${CMAKE_CURRENT_SOURCE_DIR}/book.cct)
add_test(NAME original_tree_soft_fma COMMAND cc_tree_tests_soft ${CMAKE_CURRENT_SOURCE_DIR}/startup.cct ${CMAKE_CURRENT_SOURCE_DIR}/menu.cct ${CMAKE_CURRENT_SOURCE_DIR}/book.cct)

add_executable(cc_owner_tests ../runtime010/host_owner_tests.cpp)

target_sources(cc_tree PRIVATE ../runtime010/selftest.cpp)

add_test(NAME owner_original_cases COMMAND cc_owner_tests ${CMAKE_CURRENT_SOURCE_DIR}/startup.cct ${CMAKE_CURRENT_SOURCE_DIR}/menu.cct ${CMAKE_CURRENT_SOURCE_DIR}/book.cct)
