cmake_minimum_required(VERSION 3.20)
project(cc_tree008 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(CC_SANITIZERS "Enable ASan/UBSan" OFF)
add_library(cc_tree STATIC bridge.cpp fixtures.cpp selftest.cpp ../runtime006/bridge.cpp ../xbox-runtime/guest_memory.cpp)
target_compile_options(cc_tree PUBLIC -Wall -Wextra -Werror -ffp-contract=off -fno-fast-math)
if(CC_SANITIZERS)
 target_compile_options(cc_tree PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(cc_tree PUBLIC -fsanitize=address,undefined)
endif()
add_executable(cc_tree_tests host_tests.cpp)
target_link_libraries(cc_tree_tests PRIVATE cc_tree)
enable_testing()
add_test(NAME original_tree_cases COMMAND cc_tree_tests ${CMAKE_CURRENT_SOURCE_DIR}/../renderer008/menu.cct ${CMAKE_CURRENT_SOURCE_DIR}/../renderer008/book.cct)
