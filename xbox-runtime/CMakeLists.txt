cmake_minimum_required(VERSION 3.20)
project(cc_xbox_memory_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CC_SANITIZERS "Address/undefined sanitizer host checks" ON)
add_library(cc_guest_memory guest_memory.cpp leaf_bridge.cpp)
target_compile_options(cc_guest_memory PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
add_executable(cc_bridge_tests tests.cpp)
target_link_libraries(cc_bridge_tests PRIVATE cc_guest_memory)
target_compile_options(cc_bridge_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
if(CC_SANITIZERS)
  target_compile_options(cc_guest_memory PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(cc_bridge_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cc_bridge_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME memory_bridge COMMAND cc_bridge_tests)
