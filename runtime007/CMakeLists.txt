cmake_minimum_required(VERSION 3.20)
project(cc_caller007 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(CC_SANITIZERS "ASan/UBSan host tests" OFF)
add_library(cc_caller STATIC bridge.cpp fixtures.cpp selftest.cpp ../runtime006/bridge.cpp ../xbox-runtime/guest_memory.cpp)
target_compile_options(cc_caller PUBLIC -Wall -Wextra -Werror -ffp-contract=off -fno-fast-math)
if(CC_SANITIZERS)
 target_compile_options(cc_caller PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(cc_caller PUBLIC -fsanitize=address,undefined)
endif()
add_executable(cc_caller_tests host_tests.cpp)
target_link_libraries(cc_caller_tests PRIVATE cc_caller)
enable_testing()
add_test(NAME caller_equivalence COMMAND cc_caller_tests ${CMAKE_CURRENT_SOURCE_DIR}/../renderer007/menu.ccp ${CMAKE_CURRENT_SOURCE_DIR}/../renderer007/book.ccp)

add_library(cc_fused STATIC numeric/fused.cpp)
target_compile_options(cc_fused PUBLIC -Wall -Wextra -Werror -ffp-contract=off -fno-fast-math)
if(CC_SANITIZERS)
 target_compile_options(cc_fused PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(cc_fused PUBLIC -fsanitize=address,undefined)
endif()
add_library(cc_fma_override OBJECT numeric/target_fma.cpp)
add_executable(cc_fused_tests numeric/tests.cpp)
target_link_libraries(cc_fused_tests PRIVATE cc_fused)
add_test(NAME fused_nearest COMMAND cc_fused_tests)
add_executable(cc_caller_tests_soft host_tests.cpp $<TARGET_OBJECTS:cc_fma_override>)
target_link_libraries(cc_caller_tests_soft PRIVATE cc_caller cc_fused)
add_test(NAME caller_soft_equivalence COMMAND cc_caller_tests_soft ${CMAKE_CURRENT_SOURCE_DIR}/../renderer007/menu.ccp ${CMAKE_CURRENT_SOURCE_DIR}/../renderer007/book.ccp)
